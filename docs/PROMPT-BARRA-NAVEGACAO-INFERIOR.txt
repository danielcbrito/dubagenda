interface NavItemProps {
  icon: ElementType;
  label: string;
  isActive?: boolean;
  onClick: () => void;
  badge?: number;
}

/** Item EXATO: o espaço ícone/título vem de gap-2 = 8px. */
function NavItem({ icon: Icon, label, isActive, onClick, badge }: NavItemProps) {
  return (
    <motion.button
      type="button"
      whileTap={{ scale: 0.94 }}
      onClick={onClick}
      aria-current={isActive ? "page" : undefined}
      aria-label={label}
      className="relative flex flex-1 flex-col items-center justify-start gap-2 px-1 pt-2 pb-2"
    >
      <div className="relative">
        <Icon
          aria-hidden="true"
          className={`h-[26px] w-[26px] transition-all duration-200 ${
            isActive ? "text-primary" : "text-foreground"
          }`}
          strokeWidth={isActive ? 2.2 : 1.8}
        />

        {typeof badge === "number" && badge > 0 && (
          <span className="absolute -top-1.5 -right-2 flex h-[18px] min-w-[18px] items-center justify-center rounded-full bg-primary px-1 text-[10px] font-bold leading-none tabular-nums text-primary-foreground ring-2 ring-background">
            {badge > 99 ? "99+" : badge}
          </span>
        )}
      </div>

      <span
        className={`text-[10px] leading-none tracking-wide transition-colors duration-200 ${
          isActive
            ? "font-medium text-primary"
            : "font-normal text-foreground"
        }`}
      >
        {label}
      </span>
    </motion.button>
  );
}

/** Barra inferior fixa com quatro abas de mesma largura. */
export function BottomNavigation() {

  return (

      className="fixed bottom-0 left-0 right-0 z-[60] border-t border-border/30 bg-background/95 backdrop-blur-md"

    >
      <div className="mx-auto flex h-16 w-full items-stretch justify-around">
        <NavItem
          icon={HomeIcon}
          label="Início"
          isActive={isActive("/")}
          onClick={() => navigate("/")}
        />
        <NavItem
          icon={ShoppingCart}
          label="Agenda"
          isActive={isActive("/agenda")}
          onClick={() => navigate("/agenda")}
        />

          icon={Scissors}

import { Outlet, useLocation } from "react-router-dom";
import { BottomNavigation } from "@/components/BottomNavigation";

/** Layout mobile-first com espaço reservado para a barra inferior. */
export function RootLayout() {
  const { pathname } = useLocation();

  const hideNavPatterns = [
    "/agenda/novo",
    "/disparo/novo",
    "/checkout",
    "/pagamento",
  ];

  const shouldShowNav = !hideNavPatterns.some((path) =>
    pathname.startsWith(path),
  );

  return (
    <div
      className="relative min-h-dvh w-full touch-pan-y overflow-x-hidden bg-background antialiased"
      style={
        shouldShowNav
          ? {
              paddingBottom:
                "calc(4rem + env(safe-area-inset-bottom, 0px))",
            }
          : undefined
      }
    >
      <Outlet />
      {shouldShowNav && <BottomNavigation />}
    </div>
  );
}

Se o projeto não usa Outlet, mantenha o mesmo wrapper e troque apenas
<Outlet /> por {children}. Não remova o paddingBottom condicional.

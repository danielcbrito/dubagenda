Destino: outro projeto React + Vite + TS + Tailwind + Framer Motion
Abas de destino: Início | Agenda | Disparo | Perfil

--------------------------------------------------------------
1. CONTEXTO / ESTRATÉGIA (explique isso à IA do outro projeto)
--------------------------------------------------------------
Objetivo: barra de navegação inferior com sensação de APP NATIVO
(estilo Instagram/Facebook), não de site responsivo.

Decisões de design aplicadas:
- EXATAMENTE 4 abas. Nunca 5. Sem botão central "FAB" na barra.
- Altura fixa de 64px (h-16) + padding de safe-area do iOS/Android.
- Fundo semi-transparente com blur: bg-background/95 + backdrop-blur-md.
- Borda superior sutil: border-t border-border/30 (não usar linha forte).
- Ícones OUTLINE (contorno) de 26px, nunca preenchidos por padrão.
- Cor do ícone/label INATIVO: text-foreground (PRETO real, não cinza).
  Isso foi uma decisão explícita: cinza (muted-foreground) deixava a
  barra apagada e "template de admin".
- Cor do ícone/label ATIVO: text-primary + strokeWidth 2.2 (inativo 1.8)
  e label font-medium. Sem pílula de fundo, sem indicador de linha,
  sem animação layoutId — a mudança é só de cor/peso (200ms).
- ESPAÇAMENTO ícone→label: gap-2 (8px). Foi ajustado de gap-1 para gap-2
  porque o texto ficava colado no ícone. Não reduzir.
- Label: text-[10px], leading-none, tracking-wide.
- Feedback tátil obrigatório: whileTap={{ scale: 0.94 }} (Framer Motion).
- Cada item ocupa flex-1 (larguras iguais), conteúdo alinhado ao topo
  (justify-start) com pt-2 pb-2 — evita "dança" vertical entre abas.
- Badge numérico opcional no ícone (ex.: carrinho): bolinha primary,
  18px, ring-2 ring-background para descolar do fundo, "99+" no limite.
- z-index 60 — a barra fica acima de conteúdo e de bolhas flutuantes.
- Tokens semânticos SEMPRE. Proibido text-white/bg-black/bg-gray-*.

Regras de layout que acompanham a barra:
- O container raiz da rota adiciona
  paddingBottom: calc(4rem + env(safe-area-inset-bottom, 0px))
  quando a barra está visível — nunca deixar conteúdo por baixo.
- Fluxos imersivos (agendamento/checkout/pagamento) ESCONDEM a barra.
- Qualquer botão flutuante (suporte/FAB) deve subir acima da barra:
  bottom: calc(4rem + env(safe-area-inset-bottom, 0px) + 0.875rem)

3. CÓDIGO — src/components/BottomNavigation.tsx
--------------------------------------------------------------
import { CalendarDays, Send, User } from "lucide-react";
import { motion } from "framer-motion";
import { useLocation, useNavigate } from "react-router-dom";
import type { SVGProps } from "react";

/** Ícone de casa outline (estilo Facebook) — mesma forma em todos os estados. */
function HomeIcon(props: SVGProps<SVGSVGElement>) {
  return (
    <svg viewBox="0 0 24 24" fill="none" xmlns="http://www.w3.org/2000/svg" {...props}>
      <path
        d="M12 3.4c.35 0 .69.13.96.36l7.5 6.57c.32.28.5.69.5 1.11v7.81A1.75 1.75 0 0 1 19.21 21H15.5a1 1 0 0 1-1-1v-4.5a2.5 2.5 0 1 0-5 0V20a1 1 0 0 1-1 1H4.79A1.75 1.75 0 0 1 3.04 19.25v-7.81c0-.42.18-.83.5-1.11l7.5-6.57c.27-.23.61-.36.96-.36z"
        stroke="currentColor"
        strokeWidth="1.9"
        strokeLinejoin="round"
        strokeLinecap="round"
        fill="none"
      />
    </svg>
  );
}

interface NavItemProps {
  icon: React.ElementType;
  label: string;
  isActive?: boolean;
  onClick: () => void;
  filled?: boolean;
  badge?: number;
}

function NavItem({ icon: Icon, label, isActive, onClick, filled, badge }: NavItemProps) {
  return (
    <motion.button
      whileTap={{ scale: 0.94 }}
      onClick={onClick}
      className="relative flex flex-1 flex-col items-center justify-start gap-2 px-1 pt-2 pb-2"
    >
      <div className="relative">
        <Icon
          className={`h-[26px] w-[26px] transition-all duration-200 ${
            isActive ? "text-primary" : "text-foreground"
          }`}
          strokeWidth={isActive ? 2.2 : 1.8}
          fill={filled && isActive ? "currentColor" : "none"}
        />
        {typeof badge === "number" && badge > 0 && (
          <span className="absolute -top-1.5 -right-2 min-w-[18px] h-[18px] px-1 rounded-full bg-primary text-primary-foreground text-[10px] font-bold flex items-center justify-center tabular-nums leading-none ring-2 ring-background">
            {badge > 99 ? "99+" : badge}
          </span>
        )}
      </div>
      <span
        className={`text-[10px] leading-none tracking-wide transition-colors duration-200 ${
          isActive ? "font-medium text-primary" : "font-normal text-foreground"
        }`}
      >
        {label}
      </span>
    </motion.button>
  );
}

/** Barra de navegação inferior fixa com 4 abas. */
export function BottomNavigation() {
  const location = useLocation();
  const navigate = useNavigate();

  const isActive = (path: string) => location.pathname === path;

  return (
    <nav
      className="fixed bottom-0 left-0 right-0 z-[60] border-t border-border/30 bg-background/95 backdrop-blur-md"
      style={{ paddingBottom: "env(safe-area-inset-bottom, 0px)" }}
    >
      <div className="mx-auto flex h-16 w-full items-stretch justify-around">
        <NavItem icon={HomeIcon}     label="Início"  isActive={isActive("/")}        onClick={() => navigate("/")} />
        <NavItem icon={CalendarDays} label="Agenda"  isActive={isActive("/agenda")}  onClick={() => navigate("/agenda")} />
        <NavItem icon={Send}         label="Disparo" isActive={isActive("/disparo")} onClick={() => navigate("/disparo")} />
        <NavItem icon={User}         label="Perfil"  isActive={isActive("/perfil")}  onClick={() => navigate("/perfil")} />
      </div>
    </nav>
  );
}

4. CÓDIGO — layout raiz (src/routes/RootLayout.tsx)
--------------------------------------------------------------
import { Outlet, useLocation } from "react-router-dom";
import { BottomNavigation } from "@/components/BottomNavigation";

/** Layout mobile-first com barra inferior fixa. */
export function RootLayout() {
  const { pathname } = useLocation();

  // Fluxos imersivos escondem a barra (ajuste às rotas do projeto)
  const hideNavPatterns = ["/disparo/novo", "/agenda/novo", "/checkout"];
  const shouldShowNav = !hideNavPatterns.some((p) => pathname.startsWith(p));

  return (
    <div
      className="relative min-h-dvh w-full touch-pan-y overflow-x-hidden bg-background antialiased"
      style={
        shouldShowNav
          ? { paddingBottom: "calc(4rem + env(safe-area-inset-bottom, 0px))" }
          : undefined
      }
    >
      <Outlet />
      {shouldShowNav && <BottomNavigation />}
    </div>
  );
}
